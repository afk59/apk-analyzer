# Signing a Challenge

#### Background

A challenge is a 32-byte random number used to prevent signature replay attacks. In the scenarios of [updating an enterprise public key certificate](../updating-the-enterprise-public-key-certificate/document.md) and [deleting an enterprise recovery key](../deleting-an-enterprise-recovery-key/document.md), a challenge is used to ensure operation-specific signature authorization by enterprises. The signature is generated by an enterprise using its private key of the enterprise certificate to sign the challenge based on the ECC algorithm. The challenge signature passed to the API must be a 64-byte value containing only the raw ECDSA-based signature, without any prefixes.

#### Generating a Signature for a Challenge Using the Custom Signature Tool Class **SignUtil**

To generate a signature for a challenge used by [updateEnterpriseCertificate](https://developer.huawei.com/consumer/en/doc/harmonyos-references/dataguard-recoverykey#en-us_topic_0000001983615174_section1483871653813) or [deleteEnterpriseRecoveryKey](https://developer.huawei.com/consumer/en/doc/harmonyos-references/dataguard-recoverykey#en-us_topic_0000001983615174_section1927015514386), you can utilize the custom signature tool class, where you need to replace **privateKey** and **publicKey** with the public and private keys of the enterprise, respectively.

import { cryptoFramework } from "@kit.CryptoArchitectureKit";

export class SignUtil {
  public static async signInner(data: Uint8Array) : Promise<Uint8Array> {
    // Replace the value with the private key of the enterprise.
    let privateKey: string = "-----BEGIN EC PARAMETERS-----\\n" +
      "\*\*\*\*\*\*\*\*\*\*\*\*\\n" +
      "-----END EC PARAMETERS-----\\n" +
      "-----BEGIN EC PRIVATE KEY-----\\n" +
      "\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*"  +
      "-----END EC PRIVATE KEY-----";
    // Replace the value with the public key of the enterprise.
    let publicKey: string = "-----BEGIN PUBLIC KEY-----\\n" +
      "\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\\n" +
      "\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\\n" +
      "-----END PUBLIC KEY-----\\n" +
      "-----BEGIN CERTIFICATE-----\\n" +
      "\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\\n" +
      "\*\*\*\*\*\*\*\\n" +
      "-----END CERTIFICATE-----\\n";
    let input1: cryptoFramework.DataBlob = { data };
    let signAlg = "ECC\_BrainPoolP256r1|SHA256";
    let signer = cryptoFramework.createSign(signAlg);
    let asyKeyGenerator = cryptoFramework.createAsyKeyGenerator("ECC\_BrainPoolP256r1");
    let keyPair = await asyKeyGenerator.convertPemKey(publicKey, privateKey);
    await signer.init(keyPair.priKey);
    let signData = await signer.sign(input1);
    let verifier = cryptoFramework.createVerify(signAlg);
    verifier.initSync(keyPair.pubKey);
    let res = verifier.verifySync(input1, signData);
    return signData.data;
  }

  public static async sign(data: Uint8Array) : Promise<Uint8Array> {
    let signInnerResult = await SignUtil.signInner(data);
    let result: Uint8Array = new Uint8Array(64);

    let index = 0;
    let length = 0;
    let offset = 0;
    while (index < signInnerResult.length) {
      if (signInnerResult\[index\] === 0x02) {
        length = index + 1 < signInnerResult.length ? signInnerResult\[index + 1\] : 0;
        let end = index + 2 + length;
        if (end <= signInnerResult.length) {
          let copyArr = signInnerResult.subarray(end - 32, end);
          result.set(copyArr, offset);
          offset += 32;
        }
        index += 34;
      } else {
        index++;
      }
    }
    return result;
  }
}

#### Generating a Signature for a Challenge (for Updating the Enterprise Public Key Certificate)

To update the public key certificate for an enterprise, obtain a challenge, replace **certificate** and **ecPubNewStrBase64** in the following method with the new certificate and public key, and call the **sign** method of the custom signature tool class **SignUtil** to generate the signature for the challenge.

import { util } from '@kit.ArkTS';
import { BusinessError } from '@kit.BasicServicesKit';
import { recoveryKey } from '@kit.EnterpriseDataGuardKit';
import { SignUtil } from './SignUtil';

async function updateEnterpriseCertificate() {
  // Replace the value with the new certificate of the enterprise.
  const certificate =
    "-----BEGIN CERTIFICATE-----\\n" +
      "\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\\n" +
      "\*\*\*\*\*\*\*\\n" +
      "-----END CERTIFICATE-----\\n";

  const challenge: Uint8Array = await recoveryKey.getAuthChallenge();
  const buffer = new ArrayBuffer(4);
  const view = new DataView(buffer);
  view.setUint32(0, 0x98010000);
  const command: Uint8Array = new Uint8Array(buffer);
  // Replace the value with the new public key of the enterprise.
  const ecPubNewStrBase64 =
    "\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\\n";
  let publicKey: Uint8Array = base64ToStringUint8Array(ecPubNewStrBase64);
  publicKey = publicKey.subarray(publicKey.length - 65, publicKey.length);
  let signData: Uint8Array = new Uint8Array(challenge.length + command.length + publicKey.length);
  signData.set(challenge, 0);
  signData.set(command, challenge.length);
  signData.set(publicKey, challenge.length + command.length);
  let signature: Uint8Array = await SignUtil.sign(signData);

  const cert: Uint8Array = stringToUint8(certificate!);
  recoveryKey.updateEnterpriseCertificate(signature, cert).then((ret: number) => {
    console.info(\`Succeeded in updating certificate.\`);
  }).catch((error: BusinessError) => {
    console.error(\`Failed to update certificate. Code: ${error.code}, message: ${error.message}.\`);
  });
}

function stringToUint8(str: string): Uint8Array {
  let result: Uint8Array = new Uint8Array(\[\]);
  try {
    result = new util.TextEncoder('utf-8').encodeInto(str);
  } catch (error) {
    console.error(\`Failed to encode to uint8. Code: ${error.code}, message: ${error.message}\`);
  }
  return result;
}

function base64ToStringUint8Array(base64String: string): Uint8Array {
  let base64 = new util.Base64Helper();
  let uint8Array = base64.decodeSync(base64String, util.Type.BASIC);
  return uint8Array;
}

#### Generating a Signature for a Challenge (for Deleting an Enterprise Recovery Key)

To delete an enterprise recovery key, obtain a challenge and call the **sign** method of the custom signature tool class **SignUtil** to generate the signature for the challenge.

import { BusinessError, osAccount } from '@kit.BasicServicesKit';
import { recoveryKey } from '@kit.EnterpriseDataGuardKit';
import { SignUtil } from './SignUtil';

async function deleteEnterpriseRecoveryKey() {
  const challenge: Uint8Array = await recoveryKey.getAuthChallenge();
  let signResult = await SignUtil.sign(challenge);
  let accountManager: osAccount.AccountManager = osAccount.getAccountManager();
  let userId = await accountManager.getOsAccountLocalId();
  recoveryKey.deleteEnterpriseRecoveryKey(userId, signResult).then((ret: number) => {
    console.info(\`Succeeded in deleting enterprise recovery key.\`);
  }).catch((err: BusinessError) => {
    console.error(\`Failed to delete enterprise recovery key. Code: ${err.code}, message: ${err.message}.\`);
  });
}
